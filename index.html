// TikTok処理（1080p優先取得へ修正） async function handleTikTok(rawUrl) { const parsed = parseTikTokInfo(rawUrl); try { const formData = new FormData(); formData.append('url', rawUrl); formData.append('hd', 1); // 1080pリクエスト const res = await fetch('https://www.tikwm.com/api/', { method: 'POST', body: formData }); const data = await res.json(); if (data.code === 0 && data.data) { // hdplay (1080p) が存在すれば優先、無ければ通常play (720p) を使用 const videoLink = data.data.hdplay || data.data.play; if (!videoLink) { lastErrorMessage = '[TikWM Error] 動画URLが見つかりませんでした。'; return false; } const author = (data.data.author && data.data.author.unique_id) || parsed.author || 'user'; const videoId = data.data.id || parsed.videoId || Date.now(); const fileName = `${author}(${videoId}) - TikTok.mp4`; showUserLink(author); await triggerDirectDownload(videoLink, fileName); return true; } else { lastErrorMessage = `[TikWM Error] code: ${data.code}, msg: ${data.msg || 'failed'}`; } } catch (err) { lastErrorMessage = `[TikWM Fetch Error] ${err.message}`; } return false; }
