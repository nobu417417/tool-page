// ==UserScript== // @name ダウンローダー // @namespace http://tampermonkey.net/ // @version 1.0 // @description InstagramのストーリーズURLをSaveInstaへ送って自動で解析する // @match https://www.instagram.com/* // @match https://saveinsta.app/* // @grant GM_registerMenuCommand // @grant GM_openInTab // @run-at document-end // ==/UserScript== (function() { 'use strict'; const TARGET_PAGE = 'https://saveinsta.app/ja'; // --- 1. 送信先（SaveInsta）での自動入力・実行処理 --- if (window.location.href.startsWith('https://saveinsta.app/')) { const urlParams = new URLSearchParams(window.location.search); const targetUrl = urlParams.get('url'); if (targetUrl) { const timer = setInterval(() => { const inputEl = document.getElementById('s_input'); const submitBtn = document.querySelector('button[type="submit"]') || document.querySelector('.btn-download'); if (inputEl && submitBtn) { clearInterval(timer); inputEl.value = decodeURIComponent(targetUrl); // 入力イベントを発火させてボタンを有効化 inputEl.dispatchEvent(new Event('input', { bubbles: true })); inputEl.dispatchEvent(new Event('change', { bubbles: true })); submitBtn.click(); } }, 300); } return; } // --- 2. Instagram上でのURL抽出処理 --- function getInstagramUrl() { // 現在開いているページのURL（ストーリーズ再生中のURL）を取得 return window.location.href; } // メニューコマンド登録（階層名：Tampermonkey > ダウンローダー > Instagram） GM_registerMenuCommand('Instagram', () => { const sendUrl = getInstagramUrl(); if (sendUrl && (sendUrl.includes('/stories/') || sendUrl.includes('/reels/') || sendUrl.includes('/p/'))) { const dest = `${TARGET_PAGE}?url=${encodeURIComponent(sendUrl)}`; GM_openInTab(dest, { active: true }); } else { alert('ストーリーズまたは投稿のページを開いた状態で実行してください。'); } }); })();
